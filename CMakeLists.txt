cmake_minimum_required(VERSION 3.20)
project(enigma_cardputer_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENIGMA_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(ENIGMA_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(enigma lib/enigma/Enigma.cpp)
target_include_directories(enigma PUBLIC lib/enigma)
add_library(app_model lib/app/UiModel.cpp)
target_include_directories(app_model PUBLIC lib/app)
target_link_libraries(app_model PUBLIC enigma)
add_executable(enigma_tests test/native/test_main.cpp)
target_link_libraries(enigma_tests PRIVATE enigma)
add_executable(enigma_cli test/native/cli.cpp)
target_link_libraries(enigma_cli PRIVATE enigma)
enable_testing()
add_test(NAME enigma_native COMMAND enigma_tests)
add_executable(ui_tests test/native/test_ui.cpp)
target_link_libraries(ui_tests PRIVATE app_model)
add_test(NAME ui_native COMMAND ui_tests)
